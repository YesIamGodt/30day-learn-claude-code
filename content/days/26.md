# Day 26: 特性开关

## 这一节要解决什么问题？

发布新功能，直接上线风险大。特性开关（Feature Flags）：先让 5% 的用户用到新功能，观察没问题再全量开放。

## 代码演示

```js
class FeatureFlags {
  constructor() {
    this.flags = new Map();
    this.userOverrides = new Map();
  }

  // 注册特性
  register(name, { enabled = false, rollout = 0, description = "" } = {}) {
    this.flags.set(name, { enabled, rollout, description });
  }

  // 判定是否开启
  isEnabled(flagName, userId = "anonymous") {
    // 用户级别的强制开关
    const userOverride = this.userOverrides.get(`${userId}:${flagName}`);
    if (userOverride !== undefined) return userOverride;

    const flag = this.flags.get(flagName);
    if (!flag) return false;
    if (!flag.enabled) return false;

    // 灰度 rollout：用 userId 的 hash 决定是否在范围内
    if (flag.rollout < 100) {
      const hash = this.hash(userId + flagName);
      return hash < flag.rollout;
    }

    return true;
  }

  hash(str) {
    // 简单 hash，返回 0-99 的数
    let h = 0;
    for (let i = 0; i < str.length; i++) {
      h = (Math.imul(31, h) + str.charCodeAt(i)) | 0;
    }
    return Math.abs(h) % 100;
  }

  // 强制开启/关闭（管理员用）
  setOverride(flagName, userId, enabled) {
    this.userOverrides.set(`${userId}:${flagName}`, enabled);
  }

  // 调整灰度比例
  setRollout(flagName, percent) {
    const flag = this.flags.get(flagName);
    if (flag) flag.rollout = percent;
  }

  // 获取所有特性状态
  status(userId) {
    const result = {};
    for (const [name, flag] of this.flags) {
      result[name] = {
        enabled: this.isEnabled(name, userId),
        rollout: flag.rollout,
        description: flag.description,
      };
    }
    return result;
  }
}

// 测试
const flags = new FeatureFlags();
flags.register("newREPL", { enabled: true, rollout: 30, description: "新 REPL 界面" });
flags.register("aiMemory", { enabled: true, rollout: 100, description: "AI 记忆功能" });

const users = ["alice", "bob", "charlie", "david", "alice"];
console.log("特性状态:");
for (const user of users) {
  console.log(`  ${user}: newREPL=${flags.isEnabled("newREPL", user)}, aiMemory=${flags.isEnabled("aiMemory", user)}`);
}
```
