# Day 23: 多 Agent 协作

## 这一节要解决什么问题？

一个 Agent 能力有限。多 Agent 协作：让多个专门化的 Agent 一起工作——一个负责规划，一个负责执行，一个负责审查。

## 角色分工设计

```
用户请求
    ↓
Coordinator Agent（协调者）
    ↓ 分解任务
┌───┴───┐
↓       ↓
Coder    Reviewer
Agent    Agent
↓       ↓
执行     审查
└───┬───┘
    ↓
整合结果 → 用户
```

## 代码演示

```js
// 单个 Agent
class SimpleAgent {
  constructor(name, role, systemPrompt) {
    this.name = name;
    this.role = role;
    this.systemPrompt = systemPrompt;
    this.messages = [{ role: "system", content: systemPrompt }];
  }

  async chat(message) {
    this.messages.push({ role: "user", content: message });
    // 模拟 AI 调用
    return `【${this.name}】已收到: ${message.slice(0, 50)}...`;
  }

  receive(message, from) {
    this.messages.push({ role: "user", content: `[来自${from}]: ${message}` });
  }
}

// 多 Agent 系统
class MultiAgentSystem {
  constructor() {
    this.agents = new Map();
  }

  register(agent) {
    this.agents.set(agent.name, agent);
    console.log(`✅ 注册 Agent: ${agent.name} (${agent.role})`);
  }

  // 协调者分配任务
  async coordinate(task) {
    const coordinator = this.agents.get("coordinator");
    if (!coordinator) throw new Error("缺少协调者 Agent");

    const plan = await coordinator.chat(task);
    console.log("📋 协调者计划:", plan);

    // 分发给专业 Agent
    const coder = this.agents.get("coder");
    const reviewer = this.agents.get("reviewer");

    if (coder) {
      const code = await coder.chat("根据计划写代码: " + plan);
      console.log("💻 Coder:", code.slice(0, 80));
    }

    if (reviewer) {
      const review = await reviewer.chat("审查代码");
      console.log("🔍 Reviewer:", review.slice(0, 80));
    }

    return "多 Agent 协作完成";
  }
}

// 启动
const system = new MultiAgentSystem();
system.register(new SimpleAgent("coordinator", "协调者", "你是协调者，负责分解任务。"));
system.register(new SimpleAgent("coder", "程序员", "你是资深程序员，负责写代码。"));
system.register(new SimpleAgent("reviewer", "审查员", "你是代码审查员，负责审查代码质量。"));

system.coordinate("写一个排序算法").then(r => console.log("结果:", r));
```
