# Day 11: 权限安全：什么能做，什么不能？

## 这一节要解决什么问题？

AI 能执行任意命令，这很强大，但也危险。如果没有权限控制，AI 可能删掉你的重要文件。权限系统就是给 AI 画一条"安全线"。

## 权限级别设计

```js
const PERMISSION_LEVELS = {
  NONE: 0,       // 完全禁用
  READ: 1,       // 只能读
  WRITE: 2,      // 可以写
  EXECUTE: 3,    // 可以执行命令
  DANGEROUS: 4,  // 危险操作（如删除、格式化）
};

// 权限检查
class PermissionGuard {
  constructor() {
    this.toolPermissions = new Map();
    // 默认权限：只读
    this.defaultLevel = PERMISSION_LEVELS.READ;
  }

  setPermission(toolName, level) {
    this.toolPermissions.set(toolName, level);
  }

  check(toolName, requiredLevel) {
    const level = this.toolPermissions.get(toolName) ?? this.defaultLevel;
    return level >= requiredLevel;
  }

  require(toolName, requiredLevel) {
    if (!this.check(toolName, requiredLevel)) {
      throw new Error(
        `权限不足: ${toolName} 需要级别 ${requiredLevel}，当前 ${this.toolPermissions.get(toolName) ?? this.defaultLevel}`
      );
    }
  }
}

// 使用示例
const guard = new PermissionGuard();
guard.setPermission("bash", PERMISSION_LEVELS.EXECUTE);  // 允许执行命令
guard.setPermission("file_write", PERMISSION_LEVELS.READ); // 只允许读，不允许写
guard.setPermission("bash_delete", PERMISSION_LEVELS.NONE); // 完全禁用

// 测试
console.log("bash 权限检查:", guard.check("bash", PERMISSION_LEVELS.EXECUTE)); // true
console.log("file_write 写权限检查:", guard.check("file_write", PERMISSION_LEVELS.WRITE)); // false
console.log("bash_delete 检查:", guard.check("bash_delete", PERMISSION_LEVELS.READ)); // false
```
