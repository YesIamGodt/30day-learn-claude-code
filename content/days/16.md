# Day 16: 斜杠命令：/help /exit

## 这一节要解决什么问题？

在 REPL 里输入 `/help`、`/exit` 这些特殊命令，不需要发给 AI，直接在本地处理。斜杠命令系统就是做这个的。

## 代码演示

```js
class CommandSystem {
  constructor() {
    this.commands = new Map();
  }

  // 注册命令
  register(name, handler, description = "") {
    this.commands.set(name, { handler, description });
  }

  // 解析输入，返回 { isCommand, name, args }
  parse(input) {
    if (!input.startsWith("/")) {
      return { isCommand: false, name: null, args: "" };
    }
    const [name, ...args] = input.slice(1).split(/\s+/);
    return { isCommand: true, name, args: args.join(" ") };
  }

  // 执行命令
  async execute(input) {
    const parsed = this.parse(input);
    if (!parsed.isCommand) return null;

    const cmd = this.commands.get(parsed.name);
    if (!cmd) {
      throw new Error(`未知命令: /${parsed.name}。输入 /help 查看所有命令。`);
    }

    return await cmd.handler(parsed.args);
  }

  // 显示帮助
  helpText() {
    const lines = ["可用命令:"];
    for (const [name, cmd] of this.commands) {
      lines.push(`  /${name}  - ${cmd.description}`);
    }
    return lines.join("\n");
  }
}

// 创建命令系统
const commands = new CommandSystem();

commands.register("help", () => commands.helpText(), "显示帮助");
commands.register("exit", () => { process.exit(0); }, "退出程序");
commands.register("clear", (ctx) => { ctx.history = []; return "历史已清空"; }, "清空对话历史");
commands.register("model", (ctx, args) => { ctx.model = args; return `模型切换为: ${args}`; }, "切换模型");
commands.register("cost", (ctx) => {
  const total = ctx.tokenCounter?.total() ?? 0;
  return `当前会话 Token 使用: ${total}`;
}, "显示 Token 统计");

// 测试
const ctx = { history: [], model: "claude-sonnet-4-7-20250514" };
(async () => {
  console.log(await commands.execute("/help"));
  console.log(await commands.execute("/model claude-opus-4-5"));
  console.log(await commands.execute("/unknown"));
})().catch(e => console.error(e.message));
```
