# Day 14: 用户配置：让用户自己设置

## 这一节要解决什么问题？

用户需要配置自己的 API Key、选择模型、设置偏好。配置管理系统让这些设置持久化、易修改。

## 配置加载优先级

```
命令行参数 > 环境变量 > .env 文件 > 默认值
```

## 代码演示

```js
const fs = require("fs");
const path = require("path");

class ConfigManager {
  constructor() {
    this.config = {
      apiKey: null,
      model: "claude-sonnet-4-7-20250514",
      maxTokens: 4096,
      temperature: 1.0,
      verbose: false,
    };
  }

  // 加载 .env 文件
  loadEnvFile(envPath = ".env") {
    if (!fs.existsSync(envPath)) return;

    const content = fs.readFileSync(envPath, "utf-8");
    const lines = content.split("\n");

    for (const line of lines) {
      const trimmed = line.trim();
      if (!trimmed || trimmed.startsWith("#")) continue;
      const eqIndex = trimmed.indexOf("=");
      if (eqIndex === -1) continue;

      const key = trimmed.slice(0, eqIndex).trim();
      const value = trimmed.slice(eqIndex + 1).trim().replace(/^["']|["']$/g, "");

      if (key.startsWith("ANTHROPIC_") || key.startsWith("CC_")) {
        const configKey = key.replace(/^(ANTHROPIC_|CC_)/, "").toLowerCase();
        if (configKey in this.config) {
          this.config[configKey] = value;
        }
      }
    }
  }

  // 从环境变量加载（优先级高于 .env）
  loadFromEnv() {
    if (process.env.ANTHROPIC_API_KEY) this.config.apiKey = process.env.ANTHROPIC_API_KEY;
    if (process.env.ANTHROPIC_MODEL) this.config.model = process.env.ANTHROPIC_MODEL;
    if (process.env.CC_VERBOSE === "true") this.config.verbose = true;
  }

  // 从命令行参数加载（最高优先级）
  loadFromArgs(args) {
    for (let i = 0; i < args.length; i++) {
      const arg = args[i];
      if (arg === "--api-key" && args[i + 1]) {
        this.config.apiKey = args[++i];
      }
      if (arg === "--model" && args[i + 1]) {
        this.config.model = args[++i];
      }
      if (arg === "--verbose") {
        this.config.verbose = true;
      }
    }
  }

  // 验证配置
  validate() {
    const errors = [];
    if (!this.config.apiKey) errors.push("缺少 API Key：设置 --api-key 或 ANTHROPIC_API_KEY 环境变量");
    if (!this.config.model) errors.push("缺少模型名称");
    return errors;
  }

  // 获取配置
  get(key) {
    return this.config[key];
  }

  all() {
    return { ...this.config, apiKey: this.config.apiKey ? this.config.apiKey.slice(0, 8) + "..." : null };
  }
}

// 使用示例
const config = new ConfigManager();
config.loadEnvFile();
config.loadFromEnv();
config.loadFromArgs(process.argv.slice(2));

console.log("当前配置:", config.all());
console.log("验证结果:", config.validate());
```
