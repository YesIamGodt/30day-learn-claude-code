## 代码演示

```js
// 里程碑二：组装所有组件的演示代码
// 对应 Python 源码 src/day-20/main.py 的 JavaScript 等价实现

class ClaudeCodeMilestone2 {
  constructor(apiKey) {
    // 配置系统（Day 14）
    this.config = { apiKey, model: "claude-sonnet-4-7-20250514" };

    // 任务管理器（Day 17）：追踪任务进度
    this.taskMgr = new TaskManager();
    this.taskMgr.create("实现用户登录功能");
    this.taskMgr.create("编写单元测试");
    this.taskMgr.update("1", { status: "in_progress" });

    // 计划模式（Day 18）：执行前先确认
    this.planMgr = new PlanModeManager();
    console.log(`当前模式: ${this.planMgr.mode}`);  // normal
    console.log(this.planMgr.enterPlan());          // 进入计划模式
    console.log(`切换后模式: ${this.planMgr.mode}`); // plan

    // 保存计划
    this.planMgr.savePlan("1. 分析需求 2. 编写代码 3. 测试");
    console.log(this.planMgr.exitPlan());            // 退出计划模式

    // 上下文压缩（Day 19）：处理长对话
    this.compactor = new ContextCompactor(50, 8000);

    // 权限守卫（Day 11）：危险命令检测
    this.perms = new PermissionGuard();

    // 对话历史
    this.history = new ConversationHistory();
  }

  // 模拟一轮完整的 ReAct 循环（Day 10）
  async chat(message) {
    // 1. 上下文压缩（如果历史太长）
    if (this.compactor.shouldCompact(this.history)) {
      this.compactor.compact(this.history, "已完成用户登录、单元测试相关操作");
    }

    // 2. 检查是否在 Plan 模式
    if (this.planMgr.isPlanMode()) {
      return "[计划模式] 请先退出计划模式（/exit_plan）再对话。";
    }

    // 3. 权限检查（如果涉及命令执行）
    if (!this.perms.check("chat", 1)) {
      return "[权限不足] 请检查配置。";
    }

    // 4. 正常发送消息……
    this.history.addMessage("user", message);
    // ... 调用 AI API ...
    return "模拟回复（完整组件已就绪）";
  }
}

// 运行演示
const app = new ClaudeCodeMilestone2("demo-key");
console.log("\n里程碑二组件全部就绪！");
console.log("已集成：配置、任务管理、计划模式、上下文压缩、权限守卫、历史管理");
console.log("接下来：Skills（快捷指令）→ 跨会话记忆 → 多 Agent 协作 → MCP 协议 → 插件系统");
```

## 面试题

1. **如果要在这个架构中加入"多 Agent 协作"（一个 Manager Agent 指挥多个 Worker Agent），架构上需要做哪些改动？**核心改动有两处：一是增加一个 `AgentCoordinator` 组件，负责把任务分发给多个 Worker Agent 并汇总结果；二是需要一个**消息路由层**，让来自不同 Agent 的消息能正确路由。通信方式可以是共享消息队列（同步协作）或消息总线（异步协作）。当前的单 Agent 架构主要改动的是"工具执行"部分——从单个工具变成多个 Agent 并行执行。

2. **架构中的每一个组件都可以被替换，这叫什么设计原则？有什么好处？**这叫做**依赖倒置（Dependency Inversion）**，具体体现在"面向接口编程而非面向实现编程"。每个组件（如 TaskManager、PlanModeManager）都通过明确定义的接口与外界交互，外部代码不依赖具体实现，只依赖接口。这样做的好处是：可以替换任何一个组件的实现（比如把内存存储的 TaskManager 换成数据库版本），而不影响其他代码；每个组件可以独立开发和测试。
