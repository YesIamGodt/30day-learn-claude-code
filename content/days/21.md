# Day 21: 快捷指令：把常用操作变成 /skill

## 这一节要解决什么问题？

每次让 AI "帮我代码审查"，要写一大段提示词。Skills 系统：把这些常用操作定义成快捷指令，`/review` 就搞定。

## 代码演示

```js
class Skill {
  constructor(name, description, instruction) {
    this.name = name;
    this.description = description;
    this.instruction = instruction;
  }

  // 生成发给 AI 的提示
  toPrompt(context = {}) {
    return this.instruction.replace(/\{(\w+)\}/g, (_, key) => context[key] ?? "");
  }
}

class SkillSystem {
  constructor() {
    this.skills = new Map();
  }

  register(skill) {
    this.skills.set(skill.name, skill);
  }

  // 注册内置 skills
  registerDefaults() {
    this.register(new Skill(
      "review",
      "代码审查",
      `你是一个资深代码审查员。请审查以下代码，重点关注：
1. 潜在的 bug
2. 安全漏洞
3. 代码风格
4. 性能问题

代码：{code}`
    ));

    this.register(new Skill(
      "refactor",
      "代码重构",
      `你是一个重构专家。请重构以下代码，使其更清晰、更易维护。

原则：
- 保持功能不变
- 提升可读性
- 消除重复代码

代码：{code}`
    ));

    this.register(new Skill(
      "explain",
      "代码解释",
      `你是一个耐心的老师。请用通俗易懂的语言解释以下代码。

代码：{code}`
    ));
  }

  get(name) {
    return this.skills.get(name);
  }

  list() {
    return Array.from(this.skills.values());
  }

  // 处理 /skill 命令
  execute(skillName, context) {
    const skill = this.get(skillName);
    if (!skill) throw new Error(`未知 skill: /${skillName}`);
    return skill.toPrompt(context);
  }
}

// 使用
const skills = new SkillSystem();
skills.registerDefaults();

const reviewPrompt = skills.execute("review", { code: "const x = 1" });
console.log("生成的审查提示:");
console.log(reviewPrompt);
```
