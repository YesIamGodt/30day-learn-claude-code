# Day 29: 测试与 CI

## 这一节要解决什么问题？

代码越改越多，一不小心就引入 bug。测试：每次改代码前先写测试用例，确保改动不会破坏现有功能。

## 测试金字塔

```
        /\
       /  \     E2E 测试（端到端）
      /----\    少量，耗时长
     /      \
    /   /\   \  集成测试
   /---/  \---\  较多
  /          \
 /     /\     \ 单元测试
/_____/__\_____\ 大量，速度快
```

## 代码演示

```js
// 待测试的工具类
class TokenCounter {
  count(text) {
    const 英文 = text.split(/\s+/).filter(Boolean).length;
    const 中文 = (text.match(/[\u4e00-\u9fa5]/g) || []).length;
    return Math.ceil(英文 * 1.3 + 中文);
  }

  withinBudget(text, budget) {
    return this.count(text) <= budget;
  }
}

// 简单测试框架
function test(name, fn) {
  try {
    fn();
    console.log(`✅ ${name}`);
  } catch (e) {
    console.log(`❌ ${name}`);
    console.log(`   ${e.message}`);
    return false;
  }
  return true;
}

function assertEqual(actual, expected, message = "") {
  if (actual !== expected) {
    throw new Error(`${message} 期望 ${expected}，实际 ${actual}`);
  }
}

function assertTrue(value, message = "") {
  if (!value) {
    throw new Error(`${message} 期望 true，实际 ${value}`);
  }
}

// 测试用例
const counter = new TokenCounter();
let passed = 0;
let total = 0;

total++; passed += test("英文单词 Token 计数", () => {
  assertEqual(counter.count("hello world"), 3, "两个英文单词");
});

total++; passed += test("中文 Token 计数", () => {
  const n = counter.count("你好世界");
  assertTrue(n >= 4, "4个中文");
});

total++; passed += test("混合内容", () => {
  const n = counter.count("hello 你好 world 世界");
  assertTrue(n >= 6, "混合");
});

total++; passed += test("withinBudget: 在预算内", () => {
  assertTrue(counter.withinBudget("hello", 5));
});

total++; passed += test("withinBudget: 超预算", () => {
  assertTrue(!counter.withinBudget("hello world", 1));
});

console.log(`\n📊 测试结果: ${passed}/${total} 通过`);
```
